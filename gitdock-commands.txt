# Lancer toute l'infrastructure en arrière-plan

# Lancer un seul container
docker-compose up -d gitdock-postgres
docker-compose up -d gitdock-mysql
docker-compose up -d gitdock-mongodb
docker-compose up -d gitdock-rabbitmq
docker-compose up -d gitdock-redis
docker-compose up -d gitdock-chromadb
docker-compose up -d gitdock-ollama
docker-compose up -d gitdock-pgadmin
docker-compose up -d gitdock-phpmyadmin
docker-compose up -d gitdock-mongo-express
docker-compose up -d gitdock-dozzle
docker-compose up -d gitdock-discovery

# --- ARRÊT ---
# Arrêter tous les containers (sans supprimer les volumes)
docker-compose stop

# Arrêter ET supprimer les containers (volumes conservés)
docker-compose down

# Arrêter ET supprimer containers + volumes (RESET COMPLET — perte des données)
docker-compose down -v

# --- REBUILD ---
# Rebuild sans cache et relancer (pour gitdock-discovery)
docker-compose build --no-cache gitdock-discovery
docker-compose up -d gitdock-discovery

# Rebuild tout (utile après modification d'un Dockerfile)
docker-compose build --no-cache
docker-compose up -d

# --- LOGS ---
# Logs temps réel de tous les services
docker-compose logs -f

# Logs d'un service spécifique
docker-compose logs -f gitdock-postgres
docker-compose logs -f gitdock-rabbitmq
docker-compose logs -f gitdock-discovery

# --- STATUS ---
docker-compose ps

# --- REDÉMARRAGE ---
docker-compose restart gitdock-postgres
docker-compose restart gitdock-rabbitmq
docker-compose restart gitdock-discovery

# --- ACCÈS SHELL DANS UN CONTAINER ---
docker exec -it gitdock-postgres psql -U postgres
docker exec -it gitdock-mysql mysql -u root -p
docker exec -it gitdock-rabbitmq bash
docker exec -it gitdock-redis redis-cli

13. KAFKA — COMMANDES UTILES
================================================================================

# Depuis Ubuntu (dans le container Kafka)
docker exec -it gitdock-kafka bash

# Lister les topics
kafka-topics.sh --list --bootstrap-server localhost:9092

# Décrire un topic
kafka-topics.sh --describe --topic cleaned-commits --bootstrap-server localhost:9092

# Créer un topic manuellement (si absent)
kafka-topics.sh --create --topic cleaned-commits --bootstrap-server localhost:9092 --partitions 1 --replication-factor 1

# Supprimer un topic (reset des messages)
kafka-topics.sh --delete --topic cleaned-commits --bootstrap-server localhost:9092

# Lire les messages d'un topic depuis le début
kafka-console-consumer.sh --bootstrap-server localhost:9092 --topic cleaned-commits --from-beginning

# Voir le lag du consumer group sentinel
kafka-consumer-groups.sh --bootstrap-server localhost:9092 --describe --group gitdock-sentinel-group-v4

# Reset du consumer group (forcer la relecture depuis le début)
kafka-consumer-groups.sh --bootstrap-server localhost:9092 --group gitdock-sentinel-group-v4 --reset-offsets --to-earliest --execute --topic cleaned-commits
